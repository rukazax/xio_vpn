cmake_minimum_required(VERSION 3.22)
project(xio_openvpn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/openvpn-deps")
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/mbedtls" mbedtls)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/fmt" fmt)
add_library(xio_openvpn SHARED wrapper.cpp "${DEPS}/openvpn3/client/ovpncli.cpp" "${DEPS}/openvpn3/openvpn/crypto/data_epoch.cpp" "${DEPS}/lz4/lib/lz4.c")
target_include_directories(xio_openvpn PRIVATE "${DEPS}/openvpn3" "${DEPS}/openvpn3/client" "${DEPS}/asio/include" "${DEPS}/lz4/lib")
target_compile_definitions(xio_openvpn PRIVATE ASIO_STANDALONE USE_ASIO HAVE_LZ4 USE_MBEDTLS OPENVPN_PLATFORM_ANDROID)
target_link_libraries(xio_openvpn PRIVATE mbedtls mbedx509 mbedcrypto fmt::fmt log)
target_link_options(xio_openvpn PRIVATE "-Wl,-z,max-page-size=16384")
